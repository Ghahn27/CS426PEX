MISSION calc_distance(FA speed, FA time) {
    FA distance;
    INF legs;
    CBRN label;

    distance -> speed * time;
    legs -> 2;
    label -> "leg one";

    TLP (distance > 100.0) {
        INF extra_legs;
        extra_legs -> legs + 1;
    }
}
