~ EXPECTED: PASS
~ Operations (spec 3.1, 3.2, 3.4)
~ Tests +, -, *, /, unary minus, parentheses, chaining, and order of operations
~ on INF and FA variables. Print the AST to confirm the tree shapes noted below.

MARCH {
    INF a;
    INF b;
    INF c;
    INF result;
    FA rate;
    FA total;

    a -> 10;
    b -> 4;
    c -> 2;
    rate -> 1.5;

    ~ Single operations
    result -> a + b;
    result -> a - b;
    result -> a * b;
    result -> a / b;

    ~ Chained add and subtract (left associative): ((((a + b) - c) + 7) - 1)
    result -> a + b - c + 7 - 1;

    ~ Chained multiply and divide (left associative): ((a * b) / c) * 3
    result -> a * b / c * 3;

    ~ Add/sub chained into a mult/div equation
    ~ Expected tree: (a + ((b * c) / 2)) - 3
    result -> a + b * c / 2 - 3;

    ~ Expected tree: ((a - b) + (c * a)) - (b / c)
    result -> a - b + c * a - b / c;

    ~ Parentheses override precedence
    result -> (a + b) * (c - 1) / 2;
    result -> ((a + b) * c) - (a / (b - c));

    ~ Unary minus binds tighter than * and /
    result -> -a + b;
    result -> -a * b;
    result -> -(a + b) * c;
    result -> a - -b;
    result -> --a;

    ~ Floats
    total -> rate * 2.0 + 0.5 / rate;
    total -> -rate - 1.25e2 * (rate + 3.0);
}
